# Expected fixture: wl30_sdio_enum's command sequence as docs/WIFI.md "Internal SDIO enumeration" decodes
# it, with the responses the SDIO Simplified Specification gives for the card in sdio_enum_card.txt.
# Written from the document and the specification, not from the archive. Format: tests/wifi/sdio_model.h.

# 1. CMD5 (IO_SEND_OP_COND), arg 0x1F000000, twice. R4: C set, 1 function, no memory, OCR 2.7-3.6 V
cmd5 0x1f000000 -> 0x90ff8000
cmd5 0x1f000000 -> 0x90ff8000

# 2. CMD3 (SEND_RELATIVE_ADDR): R6 offers RCA 1. CMD7 (SELECT_CARD), arg 0x00010000 = RCA 1 << 16
cmd3 0x00000000 -> 0x00010000
cmd7 0x00010000 -> ?            # R1b; the card status bits are not checked

# 3. CMD52 writes to function 0 (CCCR)
cmd52 w f0 0x00002 ff           # I/O Enable
cmd52 r f0 0x00003 02           # I/O Ready: function 1 (read before function 1 is used; not in wl30_sdio_enum as decoded)
cmd52 w f0 0x00007 02           # Bus Interface Control: 4-bit bus
cmd52 w f0 0x00004 03           # Int Enable: master and function 1

# 4. CMD52 writes of 0 to function 1 registers 0x10-0x13 (WiFi-core specific)
cmd52 w f1 0x00010 00
cmd52 w f1 0x00011 00
cmd52 w f1 0x00012 00
cmd52 w f1 0x00013 00
