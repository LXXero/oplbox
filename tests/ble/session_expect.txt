# the start: one command at a time, each after its Command Complete (ncmd 1, opcode, status 0)
cmd 0c03   # Reset (7.3.2)
evt 0e 01 03 0c 00
cmd 0c01 10 80 00 00 00 00 00 20   # Set Event Mask: Disconnection Complete, Hardware Error, LE Meta (7.3.1)

evt 0e 01 01 20 00
cmd 2002   # LE Read Buffer Size (7.8.2)
evt 0e 01 02 20 00 1b 00 02   # 27-byte payloads, 2 buffers
cmd 1009   # Read BD_ADDR (7.4.6)
evt 0e 01 09 10 00 01 00 31 4d 46 02   # 02:46:4d:31:00:01, least significant byte first
cmd 2006 30 00 30 00 00 00 00 00 00 00 00 00 00 07 00   # LE Set Advertising Parameters: 30 ms, ADV_IND, public, channels 37-39 (7.8.5)

# a central connects: LE Connection Complete (7.7.65.1): handle 0x040, peripheral, random c0:ff:ee:00:00:01, 30 ms, 720 ms

evt 13 01 40 00 01 00   # Number Of Completed Packets: one buffer back (7.7.19)
